    make package.json

    {
        "dependencies": {
            "express": "5.2.1"
        }
    }

    OR run
    npm install express after adding {}

COMMON JS
HOW TO START A NEW app.js

    const express = require('express');
    const app = express();

    app.listen(8080);


____________________________

ES MODULES
HOW TO START A NEW app.js

    import express from 'express';
    const app = express();
    
    app.listen(8080);

____________________________

ADD ERROR TO app.listen!

    const PORT = 8080;

    app.listen(PORT, (error) => {
        if (error) {
            console.log(error);
            return;
        }
        console.log("Server is running on port", PORT);
    });
