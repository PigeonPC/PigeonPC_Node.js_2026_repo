import path from 'path';

console.log(path.resolve()); //Vi får mappen på der nodemon kører fra

Vi kan få vores path 
app.get('/', (req, res) => {
    res.sendFile(path.resolve('./public/frontpage/index.html')
})

------------------------------------------------

NÅR MAN LAVER
app.get('/redirection', (req, res) => {
    res.sendFile(path.resolve('./public/redirection.html'))
})

Vi laver en GET route, på endpointet /redirection, via. en callback funktion
der håndterer en fil.

------------------------------------------------
